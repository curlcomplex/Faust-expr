cmake_minimum_required(VERSION 3.25)
project(RetainedAuthoringPublic LANGUAGES C CXX)
include(FetchContent)
find_program(NODE_EXECUTABLE node REQUIRED)
set(SNAPSHOT "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/curlop-latency")
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED}")
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(juce GIT_REPOSITORY https://github.com/curlcomplex/JUCE.git GIT_TAG 73a9390649728ba53db3229d598e88275ed166e4)
FetchContent_MakeAvailable(juce)
FetchContent_Declare(choc GIT_REPOSITORY https://github.com/curlcomplex/choc.git GIT_TAG 65a905f3d10194c95bdba049bea8256c9d313ea2 SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/vendor/choc" SOURCE_SUBDIR not-a-cmake-project)
FetchContent_MakeAvailable(choc)
find_path(FAUST_INCLUDE_DIR faust/dsp/llvm-dsp.h HINTS "${FAUST_ROOT}/include" REQUIRED)
find_library(FAUST_LIB NAMES faust libfaust HINTS "${FAUST_ROOT}/lib" REQUIRED)
file(GLOB CATALOGUE_SOURCES CONFIGURE_DEPENDS "${SNAPSHOT}/scripts/v2-language/catalogue/*.mjs")
add_custom_command(OUTPUT "${GENERATED}/ScriptV2Catalogue.generated.inc" "${GENERATED}/ScriptV2Catalogue.ts"
  COMMAND "${NODE_EXECUTABLE}" "${SNAPSHOT}/scripts/v2-language/generate-catalogue.mjs" --cpp-output "${GENERATED}/ScriptV2Catalogue.generated.inc" --ts-output "${GENERATED}/ScriptV2Catalogue.ts"
  DEPENDS "${SNAPSHOT}/scripts/v2-language/generate-catalogue.mjs" ${CATALOGUE_SOURCES} VERBATIM)
add_custom_target(catalogue DEPENDS "${GENERATED}/ScriptV2Catalogue.generated.inc")
file(READ "${SNAPSHOT}/SOURCE-MANIFEST.json" IMPORT_MANIFEST)
string(JSON SOURCE_COUNT LENGTH "${IMPORT_MANIFEST}" native_translation_units)
math(EXPR SOURCE_LAST "${SOURCE_COUNT}-1")
foreach(N RANGE 0 ${SOURCE_LAST})
  string(JSON NAME GET "${IMPORT_MANIFEST}" native_translation_units ${N})
  list(APPEND SUPPORT_SOURCES "${SNAPSHOT}/native/${NAME}")
endforeach()
function(patching_target TARGET MAIN)
  add_executable(${TARGET} "${SNAPSHOT}/scripts/bench/patching_latency/${MAIN}" ${SUPPORT_SOURCES})
  target_include_directories(${TARGET} PRIVATE "${SNAPSHOT}/native" "${GENERATED}" "${FAUST_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/vendor")
  target_compile_features(${TARGET} PRIVATE cxx_std_17)
  target_compile_options(${TARGET} PRIVATE -O2 -g -fno-omit-frame-pointer)
  target_compile_definitions(${TARGET} PRIVATE JUCE_STANDALONE_APPLICATION=1 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JucePlugin_VersionString="0.0.0-bench" CURLOP_ENABLE_FAUST_JIT=1 CURLOP_ENABLE_FAUST_RUNTIME=1 CURLOP_ENABLE_LEGACY_APG_ORACLE=0 CURLOP_FAUST_SHARE_DIR="${FAUST_ROOT}/share/faust")
  target_link_libraries(${TARGET} PRIVATE juce::juce_audio_basics juce::juce_audio_devices juce::juce_audio_processors juce::juce_audio_formats juce::juce_core juce::juce_dsp juce::juce_events "${FAUST_LIB}")
  add_dependencies(${TARGET} catalogue)
  if(APPLE)
    target_link_options(${TARGET} PRIVATE "LINKER:-stack_size,0x1000000")
  endif()
endfunction()
patching_target(RetainedPatchingBench retained_main.cpp)
