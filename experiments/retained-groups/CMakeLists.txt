cmake_minimum_required(VERSION 3.25)
project(RetainedFusedGroups LANGUAGES C CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory("../retained-host" "host")
set(SNAPSHOT "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/curlop-latency")
set(GROUP_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/group-generated")
file(MAKE_DIRECTORY "${GROUP_GENERATED}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make_runtime.py"
  "${SNAPSHOT}/scripts/bench/patching_latency/RetainedFaustGraph.h" "${GROUP_GENERATED}/retained_units.generated.h"
  RESULT_VARIABLE PATCH_RESULT)
if(NOT PATCH_RESULT EQUAL 0)
  message(FATAL_ERROR "Pinned retained runtime factory seam failed")
endif()
get_target_property(SOURCES RetainedPatchingBench SOURCES)
list(FILTER SOURCES EXCLUDE REGEX "/retained_main.cpp$")
get_target_property(INCLUDES RetainedPatchingBench INCLUDE_DIRECTORIES)
get_target_property(DEFINES RetainedPatchingBench COMPILE_DEFINITIONS)
get_target_property(LINKS RetainedPatchingBench LINK_LIBRARIES)
add_executable(StableGroupBench groups_main.cpp ${SOURCES})
target_include_directories(StableGroupBench PRIVATE ${INCLUDES} "${GROUP_GENERATED}" "${SNAPSHOT}/scripts/bench/patching_latency")
target_compile_definitions(StableGroupBench PRIVATE ${DEFINES})
target_compile_features(StableGroupBench PRIVATE cxx_std_17)
target_compile_options(StableGroupBench PRIVATE -O2 -g -fno-omit-frame-pointer)
target_link_libraries(StableGroupBench PRIVATE ${LINKS} ${CMAKE_DL_LIBS})
add_dependencies(StableGroupBench catalogue)
if(APPLE)
  target_link_options(StableGroupBench PRIVATE "LINKER:-stack_size,0x1000000")
endif()
