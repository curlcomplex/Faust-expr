cmake_minimum_required(VERSION 3.25)
project(PersistentStateNext LANGUAGES C CXX)
add_subdirectory(../persistent-state baseline)
get_target_property(SUPPORT RetainedPatchingBench SOURCES)
list(FILTER SUPPORT EXCLUDE REGEX "retained_main.cpp$")
get_target_property(INCLUDES RetainedPatchingBench INCLUDE_DIRECTORIES)
get_target_property(DEFINES RetainedPatchingBench COMPILE_DEFINITIONS)
get_target_property(LINKS RetainedPatchingBench LINK_LIBRARIES)
add_executable(PersistentStateNextBench main.generated.cpp ${SUPPORT})
target_include_directories(PersistentStateNextBench PRIVATE ${INCLUDES} .)
target_compile_definitions(PersistentStateNextBench PRIVATE ${DEFINES})
target_compile_features(PersistentStateNextBench PRIVATE cxx_std_17)
target_compile_options(PersistentStateNextBench PRIVATE -O2 -g -fno-omit-frame-pointer)
target_link_libraries(PersistentStateNextBench PRIVATE ${LINKS})
add_dependencies(PersistentStateNextBench catalogue)
if(APPLE)
 target_link_options(PersistentStateNextBench PRIVATE "LINKER:-stack_size,0x1000000")
endif()
