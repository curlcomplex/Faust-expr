cmake_minimum_required(VERSION 3.25)
project(PersistentStateExperiment LANGUAGES C CXX)
add_subdirectory(../retained-host host)
get_target_property(SUPPORT RetainedPatchingBench SOURCES)
list(FILTER SUPPORT EXCLUDE REGEX "retained_main.cpp$")
get_target_property(INCLUDES RetainedPatchingBench INCLUDE_DIRECTORIES)
get_target_property(DEFINES RetainedPatchingBench COMPILE_DEFINITIONS)
get_target_property(LINKS RetainedPatchingBench LINK_LIBRARIES)
add_executable(PersistentStateBench main.cpp ${SUPPORT})
target_include_directories(PersistentStateBench PRIVATE ${INCLUDES} .)
target_compile_definitions(PersistentStateBench PRIVATE ${DEFINES})
target_compile_features(PersistentStateBench PRIVATE cxx_std_17)
target_compile_options(PersistentStateBench PRIVATE -O2 -g -fno-omit-frame-pointer)
target_link_libraries(PersistentStateBench PRIVATE ${LINKS})
add_dependencies(PersistentStateBench catalogue)
if(APPLE)
 target_link_options(PersistentStateBench PRIVATE "LINKER:-stack_size,0x1000000")
endif()
