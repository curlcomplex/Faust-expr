cmake_minimum_required(VERSION 3.25)
project(AdaptiveRetainedGroups LANGUAGES C CXX)
add_subdirectory("../retained-groups" "previous")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ADAPTIVE_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/adaptive-generated")
file(MAKE_DIRECTORY "${ADAPTIVE_GENERATED}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make_adapter.py"
  "${CMAKE_CURRENT_SOURCE_DIR}/../retained-groups" "${ADAPTIVE_GENERATED}" RESULT_VARIABLE ADAPTER_RESULT)
if(NOT ADAPTER_RESULT EQUAL 0)
  message(FATAL_ERROR "Pinned previous-group adapters failed")
endif()
get_target_property(SOURCES StableGroupBench SOURCES)
list(FILTER SOURCES EXCLUDE REGEX "(^|/)groups_main.cpp$")
get_target_property(INCLUDES StableGroupBench INCLUDE_DIRECTORIES)
get_target_property(DEFINES StableGroupBench COMPILE_DEFINITIONS)
get_target_property(LINKS StableGroupBench LINK_LIBRARIES)
add_executable(AdaptiveGroupBench adaptive_main.cpp ${SOURCES})
target_include_directories(AdaptiveGroupBench PRIVATE ${INCLUDES} "${ADAPTIVE_GENERATED}" "${CMAKE_CURRENT_SOURCE_DIR}/../retained-groups")
target_compile_definitions(AdaptiveGroupBench PRIVATE ${DEFINES})
target_compile_features(AdaptiveGroupBench PRIVATE cxx_std_17)
target_compile_options(AdaptiveGroupBench PRIVATE -O2 -g -fno-omit-frame-pointer)
target_link_libraries(AdaptiveGroupBench PRIVATE ${LINKS})
add_dependencies(AdaptiveGroupBench catalogue)
if(APPLE)
  target_link_options(AdaptiveGroupBench PRIVATE "LINKER:-stack_size,0x1000000")
endif()
