cmake_minimum_required(VERSION 3.25)
project(AuthoringBoundaryPolicy LANGUAGES C CXX)
add_subdirectory("../evolving-groups" "previous")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(POLICY_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/policy-generated")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/make_adapter.py"
 "${CMAKE_CURRENT_SOURCE_DIR}/../evolving-groups" "${POLICY_GENERATED}" RESULT_VARIABLE ADAPTER_RESULT)
if(NOT ADAPTER_RESULT EQUAL 0)
 message(FATAL_ERROR "Pinned policy adapter failed")
endif()
get_target_property(SOURCES AdaptiveGroupBench SOURCES)
list(FILTER SOURCES EXCLUDE REGEX "(^|/)adaptive_main.cpp$")
get_target_property(INCLUDES AdaptiveGroupBench INCLUDE_DIRECTORIES)
get_target_property(DEFINES AdaptiveGroupBench COMPILE_DEFINITIONS)
get_target_property(LINKS AdaptiveGroupBench LINK_LIBRARIES)
add_executable(AuthoringPolicyBench policy_main.cpp ${SOURCES})
target_include_directories(AuthoringPolicyBench PRIVATE ${INCLUDES} "${POLICY_GENERATED}")
target_compile_definitions(AuthoringPolicyBench PRIVATE ${DEFINES})
target_compile_features(AuthoringPolicyBench PRIVATE cxx_std_17)
target_compile_options(AuthoringPolicyBench PRIVATE -O2 -g -fno-omit-frame-pointer)
target_link_libraries(AuthoringPolicyBench PRIVATE ${LINKS})
add_dependencies(AuthoringPolicyBench catalogue)
if(APPLE)
 target_link_options(AuthoringPolicyBench PRIVATE "LINKER:-stack_size,0x1000000")
endif()
